def stamp():

def field(out, key):
    m = re.search(key + r'=(\d+)', out)
    if not m:
        raise SystemExit('ABORT: 输出里找不到 %s= 这一项（汇总行格式变了？不许静默放行）' % key)
    return int(m.group(1))
