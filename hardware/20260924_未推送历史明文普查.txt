# 未推送区间的**全 blob 普查**（09-24 02:1x 第一版；02:2x 重写，理由见下面 NOTE）。
# 口令只从宏本体读出，不进命令行、不打印。
# NOTE（为什么要重写）：第一版把 `rev-list --objects` 的**全部** oid 直接喂给 `cat-file --batch`，
#   解析器只在遇到 `blob` 头时按 size 跳过内容 ⇒ 中间那些 commit/tree 对象的**内容**会被当成下一行"头"来解析，
#   解析流会错位，真实读数不可信（它给出的 396/12 属"看着合理但未证明未错位"）。
#   现改成：先 `--batch-check` 拿类型，只把 **blob** 喂进 `--batch`，并断言"解出来的只数 == 名单只数"，
#   对不上就 ABORT（宁可响，不静默给一个假"干净"）。
import os
import re
import subprocess
import sys

sys.stdout.reconfigure(encoding='utf-8')
REPO = r'C:/Users/david/Documents/all_projects/自招学习'
os.chdir(REPO)

secret = re.search(r'#define\s+PROV_PASS\s+"([^"]+)"',
                   open('hardware/zizhao-esp32s3/main/provision_ap.c', encoding='utf-8').read()).group(1).encode()
if not secret:
    print('ABORT: 宏读到了但值为空，闸无法自证扫的是真口令')
    sys.exit(1)


def objs(ref):
    """ref 可达的全部对象：oid -> 名字（名字取该 ref 下的路径，非机密）。"""
    out = subprocess.run(['git', 'rev-list', '--objects', ref], capture_output=True).stdout.decode('utf-8')
    d = {}
    for ln in out.splitlines():
        oid, _, name = ln.partition(' ')
        d[oid] = name or '(no-name)'
    return d


head = objs('HEAD')
base = objs('origin/main')
fresh = {oid: name for oid, name in head.items() if oid not in base}

# 先分类型：只扫 blob。
chk = subprocess.run(['git', 'cat-file', '--batch-check'],
                     input=b'\n'.join(k.encode() for k in fresh), capture_output=True).stdout.decode()
kinds = {}
for ln in chk.splitlines():
    parts = ln.split(' ')
    if len(parts) >= 2:
        kinds[parts[0]] = parts[1]
blob_oids = [k for k in fresh if kinds.get(k) == 'blob']
if not blob_oids:
    print('ABORT: 区间内 blob 只数为 0（候选枚举失败），不许把"0 命中"当"干净"')
    sys.exit(1)

raw = subprocess.run(['git', 'cat-file', '--batch'],
                     input=b'\n'.join(k.encode() for k in blob_oids), capture_output=True).stdout
hits = []
i = 0
parsed = 0
bad = 0
while parsed < len(blob_oids):
    j = raw.find(b'\n', i)
    if j < 0:
        bad += 1
        break
    hdr = raw[i:j].split(b' ')
    if len(hdr) >= 3 and hdr[1] == b'blob' and hdr[0].decode() == blob_oids[parsed]:
        size = int(hdr[2])
        data = raw[j + 1:j + 1 + size]
        c = data.count(secret)
        if c:
            hits.append((hdr[0].decode(), c))
        i = j + 1 + size + 1
    else:
        bad += 1
        print('ABORT: 第 %d 只解析错位，头 = %r' % (parsed, raw[i:j][:60]))
        break
    parsed += 1

print('FRESH_OBJECTS_IN_UNPUSHED_RANGE', len(fresh))
print('BLOBS_IN_RANGE', len(blob_oids))
print('BLOBS_PARSED', parsed, 'DESYNC', bad)
if bad or parsed != len(blob_oids):
    print('ABORT: 解析只数与名单只数不等 ⇒ 本轮读数作废')
    sys.exit(1)
print('BLOBS_WITH_PLAINTEXT', len(hits))
for oid, c in sorted(hits, key=lambda x: fresh[x[0]]):
    print('DIRTY %s hits=%d %s' % (oid[:8], c, fresh[oid]))
# 对照（同一把尺子，两个方向都要响）：阳性 = 上面 DIRTY 名单非空即证它会响；
# 阴性 = HEAD 那两份文档的**当前** blob 必须 0 命中，否则"树已脱敏"那句不成立。
for p in ['FreqErr.md', 'hardware/20260919_墨水屏点屏排查记录.md']:
    oid = subprocess.run(['git', 'rev-parse', 'HEAD:' + p], capture_output=True).stdout.decode().strip()
    b = subprocess.run(['git', 'cat-file', 'blob', oid], capture_output=True).stdout
    print('CONTROL_HEAD %s blob=%s hits=%d' % (p.split('/')[-1], oid[:8], b.count(secret)))
print('COMMIT_WOULD_EXPOSE_PLAINTEXT', 'YES' if hits else 'NO')

# 把"12 只脏 blob"归到具体提交（09-24 02:2x）。口令只从宏读出，不打印、不进命令行。
import os
import re
import subprocess
import sys

sys.stdout.reconfigure(encoding='utf-8')
REPO = r'C:/Users/david/Documents/all_projects/自招学习'
os.chdir(REPO)
secret = re.search(r'#define\s+PROV_PASS\s+"([^"]+)"',
                   open('hardware/zizhao-esp32s3/main/provision_ap.c', encoding='utf-8').read()).group(1).encode()
PATHS = ['FreqErr.md', 'hardware/20260919_墨水屏点屏排查记录.md',
         'hardware/zizhao-esp32s3/main/provision_ap.c']


def blob_of(ref, path):
    r = subprocess.run(['git', 'rev-parse', '--verify', '-q', ref + ':' + path], capture_output=True)
    oid = r.stdout.decode().strip()
    if not oid:
        return None, None
    b = subprocess.run(['git', 'cat-file', 'blob', oid], capture_output=True).stdout
    return oid, b.count(secret)


def added_removed(sha):
    d = subprocess.run(['git', 'show', '--format=', sha], capture_output=True).stdout
    add = rem = 0
    for ln in d.split(b'\n'):
        if ln.startswith(b'+++'):
            continue
        if ln.startswith(b'+'):
            add += ln.count(secret)
        elif ln.startswith(b'-'):
            rem += ln.count(secret)
    return add, rem


revs = subprocess.run(['git', 'rev-list', 'origin/main..HEAD'], capture_output=True).stdout.decode().split()
print('UNPUSHED_COMMITS', len(revs))
dirty_commits = []
for sha in revs:
    ts = subprocess.run(['git', 'show', '-s', '--format=%ci', sha], capture_output=True).stdout.decode().strip()
    rows = []
    tree_hit = 0
    for p in PATHS:
        oid, c = blob_of(sha, p)
        if c:
            tree_hit += 1
            rows.append('%s blob=%s hits=%d' % (p.split('/')[-1], oid[:8], c))
    add, rem = added_removed(sha)
    flag = 'DIRTY' if (tree_hit or add or rem) else 'clean'
    if flag == 'DIRTY':
        dirty_commits.append(sha[:8])
    print('%s %s %s diff(+%d/-%d) tree_dirty_paths=%d | %s' % (sha[:8], ts, flag, add, rem, tree_hit, '; '.join(rows)))
print('DIRTY_COMMITS', len(dirty_commits), ' '.join(dirty_commits))

  "读数 B"那张表 = B 的 stdout 原文，前面加了两列（sha 短码、`%ci` 提交时刻）并把 12 行按提交时刻倒序排；
  数字与判据一个都没改。B 的 `PATHS` 只有三只候选，所以它给的 `tree_dirty_paths` 是**上界口径**：
  若哪天明文进了第四只文件，B 会漏、**A 不会**（A 扫区间内全部 blob）。⇒ "A=12 / B 树脏 6 只 ×2 路径"这两句必须一起读，
  单引 B 那列数会低估。
